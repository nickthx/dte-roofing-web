## Overlap self-check script
Save this as `overlap.cjs` in your session scratchpad (NOT in the repo) and run `node <scratch>/overlap.cjs src/pages/locations/Reynoldsburg.tsx src/pages/locations/GroveCity.tsx src/pages/locations/Pickerington.tsx`. It normalises the three city names to `CITY` (so templated sentences still count as shared), builds 8-word shingles, and prints each pair's shared count and percentage of the smaller page. With `--show` as the first argument it also prints the shared shingles. Calibration on the current files: 71.8-77.9%; an unrelated page pair (Reynoldsburg vs Dublin) scores 3.8%.

```javascript
const fs = require('fs');
const args = process.argv.slice(2);
const show = args[0] === '--show';
const files = show ? args.slice(1) : args;
const norm = (s) => s.replace(/^import .*$/gm, '').replace(/className="[^"]*"/g, '').replace(/<[^>]*>/g, ' ')
  .replace(/Reynoldsburg|Grove City|Pickerington/g, 'CITY').toLowerCase().replace(/[^a-z0-9' ]+/g, ' ').split(/\s+/).filter(Boolean);
const shingles = (w) => { const S = new Set(); for (let i = 0; i + 8 <= w.length; i++) S.add(w.slice(i, i + 8).join(' ')); return S; };
const sets = files.map((f) => shingles(norm(fs.readFileSync(f, 'utf8'))));
let worst = 0;
for (let i = 0; i < files.length; i++) for (let j = i + 1; j < files.length; j++) {
  const shared = [...sets[i]].filter((x) => sets[j].has(x));
  const pct = (100 * shared.length) / Math.min(sets[i].size, sets[j].size);
  worst = Math.max(worst, pct);
  console.log(files[i].split('/').pop(), files[j].split('/').pop(), shared.length, pct.toFixed(1) + '%');
  if (show) shared.forEach((s) => console.log('   ', s));
}
process.exit(worst > 12 ? 1 : 0);
```
